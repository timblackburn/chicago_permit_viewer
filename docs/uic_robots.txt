# robots.txt — OpenAccess (shared by all clients)
#
# Goal: let search engines index records and collections, but keep them away
# from facet combinations, searches, sorting and technical endpoints, which are
# endless and expensive (every combination is a new Elasticsearch query).
#
# Only well-behaved crawlers honour it. A scraper rotating User-Agents ignores
# it: that is what the Cloudflare WAF rules and rate limiting are for.

User-agent: *
# Facets and sorting on any listing (/browse, /browse/collection/..., /search).
# limit= and offset= are NOT blocked: OpenAccess adds them to every listing,
# pagination and record link (/browse redirects to /browse?limit=10).
Disallow: /*q=facet
Disallow: /*sort=
# Views derived from /browse (facets AJAX, map, timeline)
Disallow: /browse/facets
Disallow: /browse/map
Disallow: /browse/time
Disallow: /browse/info
# Searches (endless result sets)
Disallow: /search
Disallow: /advanced_search
Disallow: /view/metadata/
Disallow: /tags
Disallow: /map
# Technical, private or expensive
Disallow: /admin
Disallow: /api/
Disallow: /auth
Disallow: /login
Disallow: /lang/
Disallow: /private
Disallow: /download/
Disallow: /download/attachment/
Disallow: /objectexplorer/
Disallow: /wayback/
# OAI-PMH harvesting and IIIF must stay reachable.
Allow: /oai
Allow: /iiif/
# Everything else (/view/, /collection/, /category/, /pages/, /thumbnail/...)
# is allowed by default. No other Allow lines on purpose: the longest match
# wins, so "Allow: /collection/" would override "Disallow: /*q=facet".
# Honoured by Bing and Yandex; Google ignores it (managed in Search Console)
